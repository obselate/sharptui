API and regression handoff

Preserve UI-thread source mutation, version invalidation, exact clipping, zero-allocation warmed emission, the bounded upload LRU, offscreen decoder hysteresis and owned process shutdown. The new MediaPlayer(path, maxSize, loop) constructor places decoder buffers and process startup at construction/restart boundaries, with explicit disposal. Maximum explicit decoding rises to 2048×2048. No consumer must coordinate upload-cache refresh or eviction. Allocation-heavy snapshots or caches must pass a new controlled comparison before API consolidation hides their cost.

Regression gates: framework selfcheck, MediaProtocolCheck.py, strict G# lint/build, generated assembly API validation, and native Discordo timeline/network/gateway/startup fixtures. Actual Ghostty rendering and input are also inspected.

func GraphicsBench() int32 {
    using let process = Process.GetCurrentProcess()
    let image = Demo.Image()
    let rgba = [640 * 320 * 4]uint8
    for i in 0 ... rgba.Length { rgba[i] = uint8(i & 255) }
    let screen = Screen(140, 46)
    let output = SharpTui.TerminalBuffer(8388608)
    for mode in []GraphicsProtocol{GraphicsProtocol.Kitty, GraphicsProtocol.Sixel} {
        screen.Graphics = mode
        for frame in 0 ... 64 {
            screen.Clear()
            screen.DrawImage(image, CellRect{Column: 32, Row: -frame % 24, WidthCells: 80, HeightRows: 20})
            screen.Emit(output)
        }
        for changing in []bool{false, true} {
            let retained = GC.GetTotalMemory(true)
            let cpu = process.TotalProcessorTime
            let before = GC.GetAllocatedBytesForCurrentThread()
            let gen0 = GC.CollectionCount(0)
            let watch = Stopwatch.StartNew()
            var emitted int64
            for frame in 0 ... 200 {
                if changing { image.Update(rgba) }
                screen.Clear()
                screen.DrawImage(image, CellRect{Column: 32, Row: -frame % 24, WidthCells: 80, HeightRows: 20})
                emitted += screen.Emit(output)
            }
            watch.Stop()
            let allocated = GC.GetAllocatedBytesForCurrentThread() - before - 40
            let cpuUsed = process.TotalProcessorTime - cpu
            Console.WriteLine(mode.ToString() + (changing ? " changing": " scrolling") + ": us/op=" + (watch.Elapsed.TotalMilliseconds * 5.0).ToString("F1") + " B/op=" + (allocated / 200).ToString() + " gen0=" + (GC.CollectionCount(0) - gen0).ToString() + " bytes/op=" + (emitted / 200).ToString() + " cpu-us/op=" + (cpuUsed.TotalMilliseconds * 5.0).ToString("F1") + " retained=" + retained.ToString())
        }
    }
    return 0
}
